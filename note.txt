public class TokenRefreshingMcpToolset extends McpToolset {

    private static final Logger logger = LoggerFactory.getLogger(TokenRefreshingMcpToolset.class);

    private final Supplier<String> tokenSupplier;
    private final String mcpUrl;
    private final ObjectMapper objectMapper;
    private volatile String currentToken;

    public TokenRefreshingMcpToolset(
            String mcpUrl,
            Supplier<String> tokenSupplier) {
        this(mcpUrl, tokenSupplier, JsonBaseModel.getMapper());
    }

    public TokenRefreshingMcpToolset(
            String mcpUrl,
            Supplier<String> tokenSupplier,
            ObjectMapper objectMapper) {
        super(buildParams(mcpUrl, tokenSupplier.get()), objectMapper);
        this.tokenSupplier = tokenSupplier;
        this.mcpUrl = mcpUrl;
        this.objectMapper = objectMapper;
        this.currentToken = tokenSupplier.get();
    }

    @Override
    public Flowable<BaseTool> getTools(ReadonlyContext readonlyContext) {
        String freshToken = tokenSupplier.get();

        if (!freshToken.equals(currentToken)) {
            logger.info("[MCP] Token changed — rebuilding MCP session for: {}", mcpUrl);
            currentToken = freshToken;
            this.close();
            rebuildSessionManager(freshToken);
        }

        return super.getTools(readonlyContext);
    }

    private void rebuildSessionManager(String newToken) {
        try {
            var field = McpToolset.class.getDeclaredField("mcpSessionManager");
            field.setAccessible(true);
            field.set(this, new McpSessionManager(buildParams(mcpUrl, newToken)));
        } catch (Exception e) {
            throw new RuntimeException("Failed to rebuild MCP session manager", e);
        }
    }

    private static StreamableHttpServerParameters buildParams(String url, String token) {
        return StreamableHttpServerParameters.builder()
            .url(url)
            .headers(Map.of("Authorization", "Bearer " + token))
            .build();
    }
}

@ApplicationScoped
@Startup
public class AtlanTokenService {

    private static final Logger logger = LoggerFactory.getLogger(AtlanTokenService.class);

    @ConfigProperty(name = "atlan.oauth.url")
    String oauthUrl;

    @ConfigProperty(name = "atlan.oauth.client-id")
    String clientId;

    @ConfigProperty(name = "atlan.oauth.client-secret")
    String clientSecret;

    private String cachedToken;
    private Instant tokenExpiry = Instant.MIN;

    @PostConstruct
    void init() {
        refreshToken();
    }

    public synchronized String getToken() {
        if (cachedToken == null || Instant.now().isAfter(tokenExpiry.minusMinutes(5))) {
            refreshToken();
        }
        return cachedToken;
    }

    private void refreshToken() {
        try {
            HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(oauthUrl))
                .header("Content-Type", "application/x-www-form-urlencoded")
                .POST(HttpRequest.BodyPublishers.ofString(
                    "grant_type=client_credentials"
                    + "&client_id=" + clientId
                    + "&client_secret=" + clientSecret))
                .build();

            HttpResponse<String> response = HttpClient.newHttpClient()
                .send(request, HttpResponse.BodyHandlers.ofString());

            var json = new ObjectMapper().readTree(response.body());
            cachedToken = json.get("access_token").asText();
            int expiresIn = json.has("expires_in") 
                ? json.get("expires_in").asInt() 
                : 3600;
            tokenExpiry = Instant.now().plusSeconds(expiresIn);

            logger.info("[TOKEN] Refreshed — expires in {}s", expiresIn);
        } catch (Exception e) {
            logger.error("[TOKEN] Refresh failed — keeping existing token", e);
        }
    }
}

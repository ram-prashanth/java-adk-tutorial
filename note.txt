public class TokenRefreshingMcpToolset extends McpToolset {

    private static final Logger logger = LoggerFactory.getLogger(TokenRefreshingMcpToolset.class);

    private final AtlanTokenService tokenService;
    private final String mcpUrl;
    private final ObjectMapper objectMapper;
    private volatile String currentToken;

    public TokenRefreshingMcpToolset(
            AtlanTokenService tokenService,
            String mcpUrl,
            ObjectMapper objectMapper) {
        // Initialize parent with current token
        super(buildParams(mcpUrl, tokenService.getToken()), objectMapper);
        this.tokenService = tokenService;
        this.mcpUrl = mcpUrl;
        this.objectMapper = objectMapper;
        this.currentToken = tokenService.getToken();
    }

    @Override
    public Flowable<BaseTool> getTools(ReadonlyContext readonlyContext) {
        String freshToken = tokenService.getToken();

        // If token changed — close old session and force parent to reinitialize
        if (!freshToken.equals(currentToken)) {
            logger.info("[MCP] Token changed — closing stale MCP session");
            currentToken = freshToken;
            this.close(); // sets mcpSession = null in parent

            // Rebuild the mcpSessionManager with fresh token by replacing it
            // McpSessionManager is stored as field in parent — we need to rebuild
            // the whole toolset via a new McpSessionManager
            rebuildSessionManager(freshToken);
        }

        return super.getTools(readonlyContext);
    }

    private void rebuildSessionManager(String newToken) {
        try {
            // McpSessionManager field is private in parent — use reflection to replace it
            var field = McpToolset.class.getDeclaredField("mcpSessionManager");
            field.setAccessible(true);
            field.set(this, new McpSessionManager(buildParams(mcpUrl, newToken)));
        } catch (Exception e) {
            throw new RuntimeException("Failed to rebuild MCP session manager", e);
        }
    }

    private static StreamableHttpServerParameters buildParams(String url, String token) {
        return StreamableHttpServerParameters.builder()
            .url(url)
            .headers(Map.of("Authorization", "Bearer " + token))
            .build();
    }
}

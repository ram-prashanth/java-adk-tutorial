package your.package.sessions;

import com.google.adk.events.Event;
import com.google.adk.sessions.BaseSessionService;
import com.google.adk.sessions.GetSessionConfig;
import com.google.adk.sessions.InMemorySessionService;
import com.google.adk.sessions.ListEventsResponse;
import com.google.adk.sessions.ListSessionsResponse;
import com.google.adk.sessions.Session;
import io.reactivex.rxjava3.core.Completable;
import io.reactivex.rxjava3.core.Maybe;
import io.reactivex.rxjava3.core.Single;
import jakarta.enterprise.context.ApplicationScoped;
import java.util.Map;
import java.util.Optional;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

@ApplicationScoped
public class A2ASessionService implements BaseSessionService {

    private static final Logger logger = LoggerFactory.getLogger(A2ASessionService.class);
    private static final String A2A_USER_PREFIX = "A2A_USER_";

    private final InMemorySessionService delegate = new InMemorySessionService();

    @Override
    public Single<Session> createSession(String appName, String userId) {
        // AgentExecutor always calls this two-arg version — intercept and fix
        // userId is always "A2A_USER_<contextId>" so extract contextId as sessionId
        String sessionId = userId.startsWith(A2A_USER_PREFIX)
            ? userId.substring(A2A_USER_PREFIX.length())
            : userId;

        logger.info("[SESSION] createSession intercepted — userId: {}, resolved sessionId: {}",
            userId, sessionId);

        return delegate.createSession(appName, userId, null, sessionId);
    }

    @Override
    public Single<Session> createSession(
            String appName, String userId,
            Map<String, Object> state, String sessionId) {
        return delegate.createSession(appName, userId, state, sessionId);
    }

    @Override
    public Maybe<Session> getSession(
            String appName, String userId,
            String sessionId, Optional<GetSessionConfig> config) {
        logger.info("[SESSION] getSession — userId: {}, sessionId: {}, instance: {}",
            userId, sessionId, System.identityHashCode(this));
        return delegate.getSession(appName, userId, sessionId, config)
            .doOnSuccess(s -> logger.info("[SESSION] HIT — eventCount: {}", s.events().size()))
            .doOnComplete(() -> logger.warn("[SESSION] MISS — userId: {}, sessionId: {}",
                userId, sessionId));
    }

    @Override
    public Single<Event> appendEvent(Session session, Event event) {
        logger.info("[SESSION] appendEvent — sessionId: {}, author: {}",
            session.id(), event.author());
        return delegate.appendEvent(session, event);
    }

    @Override
    public Completable deleteSession(String appName, String userId, String sessionId) {
        return delegate.deleteSession(appName, userId, sessionId);
    }

    @Override
    public Single<ListSessionsResponse> listSessions(String appName, String userId) {
        return delegate.listSessions(appName, userId);
    }

    @Override
    public Single<ListEventsResponse> listEvents(
            String appName, String userId, String sessionId) {
        return delegate.listEvents(appName, userId, sessionId);
    }
}
